#!/bin/bash
set -x
MODE=$1; P=/root/ngfw-wt/claude-pppoe-ipv6/.scratch/probe6; W=$P/run-$MODE
rm -rf $W; mkdir -p $W/netns
mount --bind $W/netns /run/netns
ip link set lo up
ip netns add isp
ip link add cli0 type veth peer name srv0 netns isp
ip netns exec isp sysctl -qw net.ipv6.conf.all.forwarding=1; ip link set cli0 up; ip -n isp link set lo up; ip -n isp link set srv0 up
$P/render $W/root $MODE
S=$W/root/run/ngfw/pppoe
sed -i "/^pdf=/a env > \"\$dir/env.\${reason:-x}\" 2>/dev/null || true" $W/root/etc/ppp/ngfw-dhcp6-cli0 2>/dev/null
printf "debug\nlogfile $W/dhcpcd.log\n" >> $W/root/etc/ppp/ngfw-dhcpcd-cli0.conf; mkdir -p $W/etc-ppp; cp -a /etc/ppp/. $W/etc-ppp/; cp -a $W/root/etc/ppp/. $W/etc-ppp/
printf "\"u9\" * \"probe-pw-123\" *\n" >> $W/etc-ppp/pap-secrets
mount --bind $W/etc-ppp /etc/ppp
printf "auth\nrequire-pap\nlcp-echo-interval 5\nlcp-echo-failure 3\n+ipv6\n" > $W/srv-options
ip netns exec isp pppoe-server -F -k -I srv0 -L 100.64.9.1 -R 100.64.9.10 -N 2 -O $W/srv-options > $W/server.log 2>&1 &
SRV=$!
sleep 1
pppd call ngfw-cli0 nodetach ipparam ngfw-cli0 > $W/client.log 2>&1 &
CLI=$!
for i in $(seq 30); do SIF=$(ip -n isp -o link show type ppp | awk -F": " "{print \$2}" | head -1); [ -n "$SIF" ] && break; sleep 1; done
sleep 2
ip -n isp -6 addr show dev $SIF
M=""; [ $MODE = dhcpv6 ] && M=managed
RA_PERIOD=${RA_PERIOD:-2} ip netns exec isp python3 $P/ra.py $SIF 2001:db8:9::/64 $M > $W/ra.log 2>&1 &
RA=$!
KEA=
if [ $MODE = dhcpv6 ]; then ip netns exec isp python3 $P/dhcp6s.py $SIF 2001:db8:9::100 2001:db8:9100::/56 > $W/dhcp6s.log 2>&1 & KEA=$!; fi
for i in $(seq 25); do grep -q "^gw=" $S/cli0.state6 2>/dev/null && { [ $MODE != dhcpv6 ] || grep -q "^pd=" $S/cli0.state6; } && break; sleep 1; done
echo "== after $i s"; cat $S/cli0.state6; cat $S/cli0.state; ls -la $S
ip -6 addr show dev ppp0; ip -6 route show
ps -ef | grep -E "dhcpcd|ngfw-cli0|sleep 2" | grep -v grep
GW=$(grep ^llremote= $S/cli0.state6 | cut -d= -f2)
ping -6 -c 2 -W 1 "$GW%ppp0"
LL=$(grep ^lllocal= $S/cli0.state6 | cut -d= -f2)
echo "== RS trigger check"; grep -c "rs from" $W/ra.log; ip -6 addr del "$LL" peer "$GW" dev ppp0; ip -6 addr add "$LL" peer "$GW" dev ppp0 nodad; sleep 1; grep -c "rs from" $W/ra.log; ip -6 addr show dev ppp0; ip -6 route show default
for e in $S/env.*; do echo "== $e"; grep -E "^(reason|interface|new_dhcp6_ia|new_dhcp6_ia_pd)" $e; done
kill $CLI; sleep 4
echo "== after client stop"; cat $S/cli0.state6; ls -la $S
ps -ef | grep -E "dhcpcd|ipv6-up|ngfw-cli0" | grep -v grep
kill $RA $KEA $SRV 2>/dev/null; sleep 1; pkill -f "pppoe-server -F -k -I srv0"; 
echo "== logs"; tail -30 $W/client.log; cat $W/dhcp6s.log 2>/dev/null; head -8 $W/ra.log
